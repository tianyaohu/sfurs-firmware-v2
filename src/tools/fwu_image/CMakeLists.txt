cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fwu_image C)
endif()
get_filename_component(FWU_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(fwu_image fwu_image.c ${FWU_REPO}/src/tools/lib/crc32.c)
target_include_directories(fwu_image PRIVATE ${FWU_REPO}/src/shared/inc ${FWU_REPO}/src/tools/lib)
target_compile_definitions(fwu_image PRIVATE FWU_REPO="${FWU_REPO}")
target_compile_features(fwu_image PRIVATE c_std_11)
target_compile_options(fwu_image PRIVATE -Wall -Wextra -Werror)
